Konfigurasi zones

db.febri.cfd

[root@buitenzorg ~]# cat /var/named/db.febri.cfd
$TTL 1H
@       IN SOA  xxx.febri.cfd. hostmaster.febri.cfd. (
                2025101701 ; Serial
                1H         ; Refresh
                15M        ; Retry
                1W         ; Expire
                1H )       ; Negative Cache TTL

        IN NS   xxx.febri.cfd.
        IN NS   are.fnlqxz.still.exist.febri.cfd.  ; opsional
	IN NS   8.febri.cfd.
	IN NS	febnug.febri.cfd.

febnug	IN A	103.235.75.120
8	IN A 	103.235.75.120
xxx     IN A    103.235.75.120
are.fnlqxz.still.exist    IN A    103.235.75.120  ; sementara pakai IP sama
@       IN A    103.235.75.120
www     IN A    103.235.75.120
crawler IN A    103.235.75.120
[root@buitenzorg ~]#
